

CREATE TABLE USUARIO (
    ID_USUARIO INT PRIMARY KEY AUTO_INCREMENT,
    TIPO VARCHAR(50),
    FOTO VARCHAR(255),
    BIO VARCHAR(1000),
    STATUS VARCHAR(20) DEFAULT 'ATIVO',
    DATA_CRIACAO DATE DEFAULT CURRENT_DATE,
    USERNAME VARCHAR(100),
    NOME VARCHAR(255),
    EMAIL VARCHAR(255),
    SENHA VARCHAR(255)
);



CREATE TABLE CATEGORIA (
    ID_CATEGORIA INT PRIMARY KEY AUTO_INCREMENT,
    NOME VARCHAR(100) NOT NULL UNIQUE
);



CREATE TABLE FLAG (
    ID_FLAG INT PRIMARY KEY AUTO_INCREMENT,
    NOME VARCHAR(100) NOT NULL UNIQUE
);



CREATE TABLE POSTAGEM (
    ID_POSTAGEM INT PRIMARY KEY AUTO_INCREMENT,
    ID_USUARIO INT NOT NULL,
    ID_CATEGORIA INT NOT NULL,
    ID_FLAG INT NOT NULL,
    TITULO VARCHAR(255),
    TEXTO VARCHAR(4000) NOT NULL,
    STATUS VARCHAR(50) DEFAULT 'PUBLICADO',
    DATA_POSTAGEM TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (ID_USUARIO) REFERENCES USUARIO(ID_USUARIO),
    FOREIGN KEY (ID_CATEGORIA) REFERENCES CATEGORIA(ID_CATEGORIA),
    FOREIGN KEY (ID_FLAG) REFERENCES FLAG(ID_FLAG)
);

CREATE TABLE COMENTARIO (
    ID_COMENTARIO INT PRIMARY KEY AUTO_INCREMENT,
    ID_POSTAGEM INT NOT NULL,
    ID_USUARIO INT NOT NULL,
    TEXTO VARCHAR(4000) NOT NULL,
    DATA_COMENTARIO TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (ID_POSTAGEM) REFERENCES POSTAGEM(ID_POSTAGEM),
    FOREIGN KEY (ID_USUARIO) REFERENCES USUARIO(ID_USUARIO)
);


CREATE TABLE IF NOT EXISTS PESSOA_SALVA (
    ID_USUARIO INT NOT NULL,
    ID_PESSOA INT NOT NULL,
    PRIMARY KEY (ID_USUARIO, ID_PESSOA),
    FOREIGN KEY (ID_USUARIO) REFERENCES USUARIO(ID_USUARIO),
    FOREIGN KEY (ID_PESSOA) REFERENCES USUARIO(ID_USUARIO)
);

CREATE TABLE IF NOT EXISTS PAGINA_SEGUIDA (
    ID_USUARIO INT NOT NULL,
    ID_PAGINA INT NOT NULL,
    PRIMARY KEY (ID_USUARIO, ID_PAGINA),
    FOREIGN KEY (ID_USUARIO) REFERENCES USUARIO(ID_USUARIO),
    FOREIGN KEY (ID_PAGINA) REFERENCES USUARIO(ID_USUARIO)
);


CREATE TABLE IF NOT EXISTS ITEM_SALVO (
    ID_USUARIO INT NOT NULL,
    ID_POSTAGEM INT NOT NULL,
    PRIMARY KEY (ID_USUARIO, ID_POSTAGEM),
    FOREIGN KEY (ID_USUARIO) REFERENCES USUARIO(ID_USUARIO),
    FOREIGN KEY (ID_POSTAGEM) REFERENCES POSTAGEM(ID_POSTAGEM)
);

CREATE TABLE IF NOT EXISTS FUNCIONARIO (
    ID_FUNCIONARIO INT PRIMARY KEY AUTO_INCREMENT,
    NOME VARCHAR(255) NOT NULL,
    EMAIL VARCHAR(255) NOT NULL,
    CARGO VARCHAR(50),
    STATUS VARCHAR(20) DEFAULT 'ATIVO',
    PERMISSAO VARCHAR(30)
);


CREATE TABLE IF NOT EXISTS DIARIO (
    ID_REGISTRO INT PRIMARY KEY AUTO_INCREMENT,
    ID_USUARIO INT NOT NULL,
    DATA DATE NOT NULL,
    HUMOR VARCHAR(50) NOT NULL,
    DESCRICAO VARCHAR(4000) NOT NULL,
    FOREIGN KEY (ID_USUARIO) REFERENCES USUARIO(ID_USUARIO)
);
